{{- /* Renders a small code sample with monochrome, line-based highlighting: comments, shell commands, gitjoin summary headings and list items. Usage: {{ partial "code.html" $code }} */ -}} {{- $lines := slice -}} {{- range split (strings.TrimRight "\n" .) "\n" -}} {{- $cls := "" -}} {{- $line := . | htmlEscape -}} {{- if hasPrefix . "#" -}} {{- $cls = "c-comment" -}} {{- else if hasPrefix . "$ " -}} {{- $cls = "c-cmd" -}} {{- else if hasPrefix . " - " -}} {{- $cls = "c-item" -}} {{- $line = replaceRE `(\(.*\))$` `$1` $line -}} {{- else if findRE `^[A-Z][^:]*:` . -}} {{- $cls = "c-head" -}} {{- end -}} {{- $lines = $lines | append (printf `%s` $cls $line) -}} {{- end -}} {{- printf `
%s
` (delimit $lines "\n") | safeHTML -}}
